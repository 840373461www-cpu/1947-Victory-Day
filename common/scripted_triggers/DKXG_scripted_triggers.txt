is_owned_by_ROOT_or_ally = {
	custom_trigger_tooltip = {
		tooltip = is_owned_by_ROOT_or_ally
		OWNER = {
			OR = {
				tag = ROOT
				is_subject_of = ROOT
				is_in_faction_with = ROOT
			}
		}
	}
}
